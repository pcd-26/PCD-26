% 2. Custom Date Macros
% Helper macro to convert month numbers into English text
\newcommand{\englishmonthname}[1]{%
  \ifcase#1\or January\or February\or March\or April\or May\or June\or July\or August\or September\or October\or November\or December\fi%
}

% Format: DD/MM/YYYY
\newcommand{\todayShortLittleEndian}{%
  \number\day/\number\month/\number\year%
}

% UK format: DD Month YYYY (e.g., 5 August 2026)
\newcommand{\todayFullUK}{%
  \number\day\ \englishmonthname{\number\month} \number\year%
}

% US format: Month DD, YYYY (e.g., August 5, 2026)
\newcommand{\todayFullUS}{%
  \englishmonthname{\number\month} \number\day, \number\year%
}
